% Preámbulo de LaTeX del informe (lo incluye generar.sh junto a portada.tex).
% \necesitaespacio{n\baselineskip}: salta de página si no quedan n líneas, para que tablas y código no se partan.
\usepackage{etoolbox}
\newdimen\nsFalta \newdimen\nsQueda
\newcommand{\necesitaespacio}[1]{\par\nsFalta=#1\relax\nsQueda=\pagegoal\advance\nsQueda by -\pagetotal
  \ifdim\nsFalta>\nsQueda\ifdim\nsQueda>0pt\newpage\fi\fi}
\pretocmd{\section}{\necesitaespacio{12\baselineskip}}{}{}
\pretocmd{\subsection}{\necesitaespacio{8\baselineskip}}{}{}
\usepackage{fancyvrb}
\DefineVerbatimEnvironment{Highlighting}{Verbatim}{commandchars=\\\{\},fontsize=\small}
\RecustomVerbatimEnvironment{verbatim}{Verbatim}{fontsize=\small}
% Pie de página: título y versión a la izquierda y número de página a la derecha, para que una hoja impresa suelta diga de
% qué versión es. La versión sale de version.tex, que escribe generar.sh a partir de CAMBIOS.md.
\usepackage{fancyhdr}
\fancypagestyle{plain}{\fancyhf{}\renewcommand{\headrulewidth}{0pt}
  \fancyfoot[L]{\footnotesize\color{gray}Modelos de decisión y ajuste fino · versión \informeVersion\ · \informeFecha}
  \fancyfoot[R]{\footnotesize\thepage}}
\pagestyle{plain}
% Separador de las cinco partes del informe: «PARTE II · ENTENDER» en verde, con una raya. Salta de página si no caben
% la parte y el principio de su primera sección. (No se usa \part, que choca con \necesitaespacio.)
\newcounter{parteinforme}
\newcommand{\parteinforme}[1]{\par\necesitaespacio{16\baselineskip}\stepcounter{parteinforme}\vspace{1.2em}%
  {\color[HTML]{5F8A12}\small\bfseries\addfontfeatures{LetterSpace=12}PARTE \Roman{parteinforme} · \MakeUppercase{#1}}\par
